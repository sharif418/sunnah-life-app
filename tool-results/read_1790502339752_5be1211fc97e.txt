    23→function dateKey(d: Date): string {
    24→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    25→}
    26→function parseKey(key: string): Date {
    27→  const [y, m, d] = key.split("-").map(Number);
    28→  return new Date(y, m - 1, d);
    29→}
    30→function addDays(key: string, days: number): string {
    31→  const d = parseKey(key);
    32→  d.setDate(d.getDate() + days);
    33→  return dateKey(d);
    34→}
    35→function weekStartOfSafe(d: Date): string {
    36→  const diff = (d.getDay() + 1) % 7; // Sat → 0, Sun → 1, … Fri → 6
    37→  return dateKey(new Date(d.getFullYear(), d.getMonth(), d.getDate() - diff));
    38→}
    39→function hijriArithmetic(d: Date): { year: number; monthIndex: number; day: number } {
    40→  const jd = Math.floor(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()) / 86400000) + 2440588;
    41→  let l = jd - 1948440 + 10632;
    42→  const n = Math.floor((l - 1) / 10631);
    43→  l = l - 10631 * n + 354;
    44→  const j =
    45→    Math.floor((10985 - l) / 5316) * Math.floor((50 * l) / 17719) +
    46→    Math.floor(l / 5670) * Math.floor((43 * l) / 15238);
    47→  l =
    48→    l -
    49→    Math.floor((30 - j) / 15) * Math.floor((17719 * j) / 50) -
    50→    Math.floor(j / 16) * Math.floor((15238 * j) / 43) +
    51→    29;
    52→  const month = Math.floor((24 * l) / 709);
    53→  const day = l - Math.floor((709 * month) / 24);
    54→  const year = 30 * n + j - 30;
    55→  return { year, monthIndex: month - 1, day };
    56→}
    57→function hijriIso(d: Date): string {
    58→  const h = hijriArithmetic(d);
    59→  return `${h.year}-${h.monthIndex + 1}-${h.day}`;
    60→}
    61→function isAyyamBeez(d: Date): boolean {
    62→  const day = hijriArithmetic(d).day;
    63→  return day === 13 || day === 14 || day === 15;

    66→// deterministic PRNG so reseeding always produces the same history
    67→function mulberry32(seed: number) {
    68→  let a = seed >>> 0;
    69→  return () => {
    70→    a |= 0;
    71→    a = (a + 0x6d2b79f5) | 0;
    72→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    73→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    74→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    75→  };
    76→}
    77→function hashStr(s: string): number {
    78→  let h = 2166136261;
    79→  for (const c of s) h = Math.imul(h ^ c.charCodeAt(0), 16777619);
    80→  return h >>> 0;
    81→}
