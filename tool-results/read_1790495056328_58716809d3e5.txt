    22→    22→function mulberry32(seed: number) {
    23→    23→  let a = seed >>> 0;
    24→    24→  return () => {
    25→    25→    a |= 0;
    26→    26→    a = (a + 0x6d2b79f5) | 0;
    27→    27→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    28→    28→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    29→    29→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    30→    30→  };
    31→    31→}
